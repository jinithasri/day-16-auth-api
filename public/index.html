//const API = ""; // ===================================================== // SHOW MESSAGE // ===================================================== function showMessage(message) { document.getElementById("message").innerText = message; } // ===================================================== // GET JWT TOKEN // ===================================================== function getToken() { return localStorage.getItem("token"); } // ===================================================== // CHECK LOGIN // ===================================================== function isLoggedIn() { return !!getToken(); } // ===================================================== // AUTH HEADERS // ===================================================== function authHeaders() { const token = getToken(); return { Authorization: `Bearer ${token}` }; } // ===================================================== // SIGNUP // ===================================================== async function signup() { try { const name = document.getElementById("signupName").value.trim(); const email = document.getElementById("signupEmail").value.trim(); const password = document.getElementById("signupPassword").value; if (!name || !email || !password) { showMessage( "Please fill all signup fields" ); return; } const response = await axios.post( `${API}/signup`, { name, email, password } ); showMessage( response.data.message ); document.getElementById("signupName").value = ""; document.getElementById("signupEmail").value = ""; document.getElementById("signupPassword").value = ""; } catch (error) { console.error("Signup error:", error); showMessage( error.response?.data?.message || "Signup failed" ); } } // ===================================================== // LOGIN // ===================================================== async function login() { try { const email = document.getElementById("loginEmail").value.trim(); const password = document.getElementById("loginPassword").value; if (!email || !password) { showMessage( "Please enter email and password" ); return; } const response = await axios.post( `${API}/login`, { email, password } ); // Save JWT token localStorage.setItem( "token", response.data.token ); showMessage( "Login successful!" ); // Automatically load users await getUsers(); } catch (error) { console.error("Login error:", error); showMessage( error.response?.data?.message || "Login failed" ); } } // ===================================================== // LOGOUT // ===================================================== async function logout() { try { const token = getToken(); if (!token) { showMessage( "Already logged out" ); return; } await axios.post( `${API}/logout`, {}, { headers: authHeaders() } ); localStorage.removeItem("token"); document.getElementById( "userTable" ).innerHTML = ""; document.getElementById( "profile" ).innerText = ""; showMessage( "Logged out successfully" ); } catch (error) { console.error("Logout error:", error); // Remove token anyway localStorage.removeItem("token"); document.getElementById( "userTable" ).innerHTML = ""; document.getElementById( "profile" ).innerText = ""; showMessage( "Logged out successfully" ); } } // ===================================================== // GET PROFILE // ===================================================== async function getProfile() { try { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const response = await axios.get( `${API}/profile`, { headers: authHeaders() } ); document.getElementById( "profile" ).innerText = JSON.stringify( response.data, null, 2 ); showMessage( "Profile loaded" ); } catch (error) { console.error( "Profile error:", error.response?.data || error ); if ( error.response?.status === 401 || error.response?.status === 400 ) { localStorage.removeItem("token"); } showMessage( error.response?.data?.message || "Could not fetch profile" ); } } // ===================================================== // GET ALL USERS // ===================================================== async function getUsers() { try { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const response = await axios.get( `${API}/users`, { headers: authHeaders() } ); displayUsers( response.data ); showMessage( `${response.data.length} user(s) loaded` ); } catch (error) { console.error( "Get users error:", error.response?.data || error ); if ( error.response?.status === 401 || error.response?.status === 400 ) { localStorage.removeItem("token"); } showMessage( error.response?.data?.message || "Could not fetch users" ); } } // ===================================================== // DISPLAY USERS // ===================================================== function displayUsers(users) { const table = document.getElementById("userTable"); table.innerHTML = ""; if (!users || users.length === 0) { table.innerHTML = ` No users found `; return; } users.forEach(user => { const row = document.createElement("tr"); row.innerHTML = ` ${user.name || ""} ${user.email || ""} ${user.age || ""} `; table.appendChild(row); }); } // ===================================================== // ADD USER // ===================================================== async function addUser() { try { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const name = document.getElementById( "userName" ).value.trim(); const email = document.getElementById( "userEmail" ).value.trim(); const age = Number( document.getElementById( "userAge" ).value ); if (!name || !email || !age) { showMessage( "Please enter name, email and age" ); return; } await axios.post( `${API}/users`, { name, email, age }, { headers: authHeaders() } ); showMessage( "User added successfully" ); document.getElementById( "userName" ).value = ""; document.getElementById( "userEmail" ).value = ""; document.getElementById( "userAge" ).value = ""; await getUsers(); } catch (error) { console.error( "Add user error:", error.response?.data || error ); showMessage( error.response?.data?.message || "Could not add user" ); } } // ===================================================== // EDIT USER // ===================================================== async function editUser(id) { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const name = prompt( "Enter new name:" ); if (!name) { return; } try { await axios.put( `${API}/users/${id}`, { name }, { headers: authHeaders() } ); showMessage( "User updated successfully" ); await getUsers(); } catch (error) { console.error( "Edit user error:", error.response?.data || error ); showMessage( error.response?.data?.message || "Could not update user" ); } } // ===================================================== // DELETE USER // ===================================================== async function deleteUser(id) { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const confirmed = confirm( "Delete this user?" ); if (!confirmed) { return; } try { await axios.delete( `${API}/users/${id}`, { headers: authHeaders() } ); showMessage( "User deleted successfully" ); await getUsers(); } catch (error) { console.error( "Delete user error:", error.response?.data || error ); showMessage( error.response?.data?.message || "Could not delete user" ); } } // ===================================================== // SEARCH USERS // ===================================================== async function searchUsers() { try { if (!isLoggedIn()) { showMessage( "Please login first" ); return; } const search = document.getElementById( "searchInput" ).value .toLowerCase() .trim(); if (!search) { await getUsers(); return; } // Get all users from backend const response = await axios.get( `${API}/users`, { headers: authHeaders() } ); // Filter users by name const filtered = response.data.filter( user => user.name && user.name .toLowerCase() .includes(search) ); displayUsers( filtered ); showMessage( `${filtered.length} user(s) found` ); } catch (error) { console.error( "Search error:", error.response?.data || error ); showMessage( error.response?.data?.message || "Search failed" ); } }
